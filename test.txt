package main

import (
	"fmt"
	"time"
)

func main() {

	host := "www.google.com"
	hop := RunTraceroute(host)

	for _, h := range hop {

		if len(hop) == 0 {

			continue
		}

		total := time.Duration(0)
		for _, rtt := range h.RTTs {
			total += rtt
		}
		avgRTT := total / time.Duration(len(h.RTTs))
		fmt.Printf("Average RTT: %v\n", avgRTT)

	}

}

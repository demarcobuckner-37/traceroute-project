package main

import (
	"time"
)

func main() {

	host := "www.google.com"
	hopA := RunTraceroute(host, Fault{Enabled: false})
	hopA = RunAnalyzeHops(hopA)
	RunBottleneckDetection(hopA)
	RouteSummary(hopA)
	PrintHopAnalysis(hopA)

	hopB := RunTraceroute(host, Fault{Enabled: true, DropRate: 0.3, Delay: 100 * time.Millisecond})
	hopB = RunAnalyzeHops(hopB)
	RunBottleneckDetection(hopB)
	RouteSummary(hopB)
	PrintHopAnalysis(hopB)

	RunRouteComparison(hopA, hopB)

}
